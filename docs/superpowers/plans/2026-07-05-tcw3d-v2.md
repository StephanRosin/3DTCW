- [ ] **Step 1: `js/textures.js` schreiben**

```javascript
import * as THREE from 'three';

const loader = new THREE.TextureLoader();

export function loadTex(path, { repeat = [1, 1], srgb = false } = {}) {
  const tex = loader.load(path, undefined, undefined,
    () => console.warn(`Textur fehlt: ${path} — Fallback-Farbe aktiv`));
  tex.wrapS = tex.wrapT = THREE.RepeatWrapping;
  tex.repeat.set(repeat[0], repeat[1]);
  tex.anisotropy = 8;
  if (srgb) tex.colorSpace = THREE.SRGBColorSpace;
  return tex;
}

/** PBR-Material aus einem Textur-Set-Ordner (diff.jpg [+ nor.jpg]). */
export function pbr({ dir, color = 0xffffff, repeat = [1, 1], roughness = 1, metalness = 0, normalScale = 1 }) {
  const mat = new THREE.MeshStandardMaterial({ color, roughness, metalness });
  mat.map = loadTex(`${dir}/diff.jpg`, { repeat, srgb: true });
  // Normal-Map nur setzen, wenn Datei existiert — Existenz zur Ladezeit unbekannt,
  // deshalb optimistisch setzen; onError entfernt sie wieder.
  const nor = loader.load(`${dir}/nor.jpg`,
    () => { mat.normalMap = nor; mat.normalScale.set(normalScale, normalScale); mat.needsUpdate = true; },
    undefined,
    () => {});
  nor.wrapS = nor.wrapT = THREE.RepeatWrapping;
  nor.repeat.set(repeat[0], repeat[1]);
  return mat;
}
```

```javascript
export const ROW_COURTS = 6;
export const ENC = { minX: -48.8, maxX: 48.8, minZ: -18, maxZ: 18, h: 4 };

export function buildCourtRow(scene) {
  const group = new THREE.Group();
  scene.add(group);

  // Eine durchgehende Sandfläche über die ganze Anlage (wie auf den Luftbildern).
  const clay = new THREE.Mesh(
    new THREE.PlaneGeometry(ENC.maxX - ENC.minX, ENC.maxZ - ENC.minZ),
    pbr({ dir: 'assets/textures/clay', color: 0xb0532e, repeat: [24, 9] }));
  clay.rotation.x = -Math.PI / 2;
  clay.position.y = 0.02;
  clay.receiveShadow = true;
  group.add(clay);

  const courtX = [];
  for (let i = 0; i < ROW_COURTS; i++) {
    const cx = (i - (ROW_COURTS - 1) / 2) * COURT_PITCH;
    courtX.push(cx);
    addCourtMarkings(group, cx);   // Linien-Plane + Netz + Netzpfosten (aus Bestandscode)
  }

  buildEnclosureFence(group);      // Aussenzaun 4 m: unten 2.2 m Windschutz, oben Maschendraht
  addFenceColliders();             // Kollision mit 2 Tor-Lücken auf Nordseite (x=-10, x=+40)

  for (const x of [-COURT_PITCH * 2, 0, COURT_PITCH * 2]) {
    buildFloodlight(scene, x, ENC.minZ - 0.8, 14);
    buildFloodlight(scene, x, ENC.maxZ + 0.8, 14);
  }
  return { courtX };
}
```

- [ ] **Step 1: `groundHeight` implementieren**

```javascript
export const PLATEAU = { minX: -28, maxX: 8, minZ: -29, maxZ: -19, h: 0.75, stepDepth: 1.2 };

export function groundHeight(x, z) {
  const p = PLATEAU;
  if (x < p.minX || x > p.maxX) return 0;
  if (z >= p.minZ && z <= p.maxZ) return p.h;
  if (z > p.maxZ && z <= p.maxZ + p.stepDepth) {
    return p.h * (1 - (z - p.maxZ) / p.stepDepth);
  }
  return 0;
}
```

```javascript
import * as THREE from 'three';

export function createInfoPoints(scene, camera, defs) {
  const hint = document.getElementById('infohint');
  const card = document.getElementById('infocard');
  const sprites = defs.map((d) => {
    const c = document.createElement('canvas');
    c.width = c.height = 128;
    const g = c.getContext('2d');
    g.fillStyle = '#26418f'; g.beginPath(); g.arc(64, 64, 56, 0, Math.PI * 2); g.fill();
    g.strokeStyle = '#fff'; g.lineWidth = 6; g.stroke();
    g.fillStyle = '#fff'; g.font = 'bold italic 84px Georgia';
    g.textAlign = 'center'; g.textBaseline = 'middle'; g.fillText('i', 64, 68);
    const tex = new THREE.CanvasTexture(c); tex.colorSpace = THREE.SRGBColorSpace;
    const s = new THREE.Sprite(new THREE.SpriteMaterial({ map: tex, depthTest: true }));
    s.scale.set(0.45, 0.45, 1);
    s.position.set(d.x, d.y ?? 1.6, d.z);
    scene.add(s);
    return { sprite: s, def: d, baseY: d.y ?? 1.6 };
  });

  let open = false, near = null, t = 0;
  window.addEventListener('keydown', (e) => {
    if (e.code !== 'KeyE' || !near) return;
    open = !open;
    if (open) {
      card.querySelector('h2').textContent = near.def.title;
      card.querySelector('p').textContent = near.def.text;
    }
    card.classList.toggle('visible', open);
  });

  function update(dt) {
    t += dt;
    near = null;
    let best = 9;   // = 3m²
    for (const p of sprites) {
      p.sprite.position.y = p.baseY + Math.sin(t * 2 + p.baseY) * 0.1;
      const dx = camera.position.x - p.def.x, dz = camera.position.z - p.def.z;
      const d2 = dx * dx + dz * dz;
      if (d2 < best) { best = d2; near = p; }
    }
    hint.classList.toggle('visible', !!near && !open);
    if (!near && open) { open = false; card.classList.remove('visible'); }
  }
  return { update };
}
```
